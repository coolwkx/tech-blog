```python
# -*- coding: utf-8 -*-
"""GBDT 演示：初始化均值 -> 拟合残差 -> 找最优切分点"""
import numpy as np

x = np.arange(1, 11)
y = np.array([5.56, 5.70, 5.91, 6.40, 6.80, 7.05, 8.90, 8.70, 9.00, 9.05])

# Step 1 初始化：平方损失下的最优常数 = 均值
f0 = y.mean
print(f"f0 = {f0:.4f}")

# Step 2 负梯度（平方损失时就是残差）
r = y - f0
print("残差:", np.round(r, 4))

# Step 3 用残差作为新目标，找平方损失最小的切分点
best = None
for s in np.arange(1.5, 10, 1.0):
    L, R = r[x < s], r[x >= s]
    m = ((L - L.mean) ** 2).sum + ((R - R.mean) ** 2).sum
    print(f"s={s:4.1f} 左均值={L.mean:7.4f} 右均值={R.mean:7.4f} m(s)={m:8.4f}")
    if best is None or m < best[1]:
        best = (s, m, L.mean, R.mean)

        print(f"\n第1棵树最优切分点 s={best[0]}, m={best[1]:.4f}, 左输出={best[2]:.4f}, 右输出={best[3]:.4f}")

```
